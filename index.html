const express = require('express'); const app = express(); app.use(express.json()); // Inscription et détection du mot de passe admin app.post('/api/auth/register', async (req, res) => { const { fullName, email, phone, address, password } = req.body; const ADMIN_KEY = process.env.ADMIN_SECRET_KEY || "winner@2"; if (password === ADMIN_KEY) { return res.json({ success: true, role: 'ADMIN_AGENT', message: "Bienvenue dans le panneau d'administration." }); } else { return res.json({ success: true, role: 'USER', requireOTP: true, message: "Un code de confirmation vous a été envoyé par SMS." }); } }); app.listen(3000, () => console.log('Serveur démarré sur le port 3000'));
